cmake_minimum_required(VERSION 3.16)
project(sarc_codec LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_library(sarc_codec src/sarc_codec.cpp)
target_include_directories(sarc_codec PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/include)
target_compile_options(sarc_codec PRIVATE -Wall -Wextra -Wpedantic)
add_executable(sarc_codec_tests tests/test_codec.cpp)
target_link_libraries(sarc_codec_tests PRIVATE sarc_codec)
enable_testing()
add_test(NAME codec_roundtrip COMMAND sarc_codec_tests)
